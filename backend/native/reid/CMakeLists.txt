cmake_minimum_required(VERSION 3.18)
project(reid_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ONNXRUNTIME_INCLUDE_DIR "" CACHE PATH "Directory containing onnxruntime_cxx_api.h")
if(NOT EXISTS "${ONNXRUNTIME_INCLUDE_DIR}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set ONNXRUNTIME_INCLUDE_DIR to ONNX Runtime SDK include directory (>=1.20)")
endif()
add_library(reid_runtime SHARED src/reid_runtime.cpp)
set_target_properties(reid_runtime PROPERTIES OUTPUT_NAME reid_runtime_v2)
target_include_directories(reid_runtime PRIVATE include "${ONNXRUNTIME_INCLUDE_DIR}")
if(MSVC)
  target_compile_options(reid_runtime PRIVATE /O2 /EHsc /utf-8)
else()
  target_compile_options(reid_runtime PRIVATE -O3 -fvisibility=hidden)
  target_link_libraries(reid_runtime PRIVATE ${CMAKE_DL_LIBS})
endif()
install(TARGETS reid_runtime RUNTIME DESTINATION . LIBRARY DESTINATION .)
